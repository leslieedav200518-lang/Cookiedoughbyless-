    <script>
        // Estructura de Datos y Estado Local
        let clients = JSON.parse(localStorage.getItem('cookie_clients')) || [];
        let currentClientIndex = clients.length > 0 ? 0 : null;
        let pendingAction = null; // 'stamp' o 'edit'

        // Datos iniciales de prueba si no hay nada
        if (clients.length === 0) {
            const sampleClient = {
                id: 'CDL-' + Math.floor(1000 + Math.random() * 9000),
                name: 'Mariana López',
                date: new Date().toLocaleDateString('es-MX'),
                services: 4,
                history: ['02/10/2026', '04/10/2026', '05/10/2026', '06/10/2026']
            };
            clients.push(sampleClient);
            currentClientIndex = 0;
            saveData();
        }

        function saveData() {
            localStorage.setItem('cookie_clients', JSON.stringify(clients));
        }

        // Renderizado de la tarjeta actual
        function renderCard() {
            if (currentClientIndex === null || clients.length === 0) return;
            const client = clients[currentClientIndex];

            document.getElementById('clientNameDisp').innerText = client.name;
            document.getElementById('clientDateDisp').innerText = `Creada el: ${client.date}`;
            document.getElementById('clientIDDisp').innerText = `ID: ${client.id}`;
            document.getElementById('progressCounter').innerText = `Progreso: ${client.services} de 10 servicios`;

            // Mensaje de celebración si llega a 10
            const celebMsg = document.getElementById('celebrationMsg');
            if (client.services >= 10) {
                celebMsg.style.display = 'block';
            } else {
                celebMsg.style.display = 'none';
            }

            // Renderizar los 10 espacios de galletas
            const grid = document.getElementById('cookiesGrid');
            grid.innerHTML = '';
            for (let i = 1; i <= 10; i++) {
                const slot = document.createElement('div');
                slot.className = 'cookie-slot';
                if (i <= client.services) {
                    slot.classList.add('active');
                    slot.innerHTML = '🍪';
                } else {
                    slot.innerHTML = '🍪';
                }
                grid.appendChild(slot);
            }

            // Renderizar Historial
            const historyList = document.getElementById('historyList');
            historyList.innerHTML = '';
            if (client.history.length === 0) {
                historyList.innerHTML = '<li>Sin compras registradas aún.</li>';
            } else {
                client.history.forEach(date => {
                    const li = document.createElement('li');
                    li.innerText = `Compra registrada el: ${date}`;
                    historyList.appendChild(li);
                });
            }
        }

        // Sistema de PIN Administrador
        function requestAdminAction(action, param = null) {
            pendingAction = { type: action, param: param };
            document.getElementById('pinInput').value = '';
            document.getElementById('pinModal').style.display = 'flex';
            document.getElementById('pinInput').focus();
        }

        function closePinModal() {
            document.getElementById('pinModal').style.display = 'none';
            pendingAction = null;
        }

        function verifyPin() {
            const pin = document.getElementById('pinInput').value;
            if (pin === '2004') {
                closePinModal();
                executePendingAction();
            } else {
                alert('PIN incorrecto. Intenta de nuevo.');
                document.getElementById('pinInput').value = '';
            }
        }

        function executePendingAction() {
            if (!pendingAction) return;

            if (pendingAction.type === 'stamp') {
                addServiceToCurrent();
            } else if (pendingAction.type === 'edit') {
                editClientName(pendingAction.param);
            }
        }

        // Marcar Servicio (Sello digital)
        function addServiceToCurrent() {
            const client = clients[currentClientIndex];
            if (client.services >= 10) {
                if (confirm('Esta tarjeta ya completó 10 servicios. ¿Deseas reiniciar la tarjeta para iniciar un nuevo ciclo?')) {
                    client.services = 0;
                    client.history = [];
                    saveData();
                    renderCard();
                    return;
                } else {
                    return;
                }
            }

            client.services += 1;
            const today = new Date().toLocaleDateString('es-MX');
            client.history.push(today);
            saveData();
            renderCard();

            if (client.services === 10) {
                alert('¡Felicidades hermosa! Has alcanzado tu recompensa 🍪✨ (1 galleta clásica gratis)');
            } else if (client.services === 5) {
                alert('¡Genial! La clienta ha alcanzado el Servicio 5 (10% de descuento en paquete de 6 galletas)');
            }
        }

        // Crear Nueva Tarjeta
        function promptNewCard() {
            document.getElementById('newClientNameInput').value = '';
            document.getElementById('newCardModal').style.display = 'flex';
            document.getElementById('newClientNameInput').focus();
        }

        function closeNewCardModal() {
            document.getElementById('newCardModal').style.display = 'none';
        }

        function createNewCard() {
            const name = document.getElementById('newClientNameInput').value.trim();
            if (!name) {
                alert('Por favor ingresa el nombre de la clienta.');
                return;
            }

            const newClient = {
                id: 'CDL-' + Math.floor(1000 + Math.random() * 9000),
                name: name,
                date: new Date().toLocaleDateString('es-MX'),
                services: 0,
                history: []
            };

            clients.unshift(newClient);
            currentClientIndex = 0;
            saveData();
            closeNewCardModal();
            renderCard();
            if(document.getElementById('adminControls').style.display === 'block') {
                filterClients();
            }
        }

        // Panel de Gestión de Clientes (Administrador)
        function toggleAdminSearch() {
            // Requiere PIN para abrir la lista de clientes
            requestAdminAction('open_admin_list');
        }

        // Sobrescribimos temporalmente el manejador para la acción de abrir lista
        const oldVerifyPin = verifyPin;
        verifyPin = function() {
            const pin = document.getElementById('pinInput').value;
            if (pin === '2004') {
                closePinModal();
                if (pendingAction && pendingAction.type === 'open_admin_list') {
                    const controls = document.getElementById('adminControls');
                    controls.style.display = controls.style.display === 'none' ? 'block' : 'none';
                    if (controls.style.display === 'block') {
                        filterClients();
                    }
                } else {
                    executePendingAction();
                }
            } else {
                alert('PIN incorrecto. Intenta de nuevo.');
                document.getElementById('pinInput').value = '';
            }
        };

        function filterClients() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const container = document.getElementById('clientListContainer');
            container.innerHTML = '';

            const filtered = clients.filter(c => c.name.toLowerCase().includes(query) || c.id.toLowerCase().includes(query));

            if (filtered.length === 0) {
                container.innerHTML = '<p style="text-align:center; color: var(--sec-2); font-size:0.85rem; padding:10px;">No se encontraron clientas.</p>';
                return;
            }

            filtered.forEach(client => {
                const realIndex = clients.findIndex(c => c.id === client.id);
                const item = document.createElement('div');
                item.className = 'client-item';
                item.innerHTML = `
                    <div>
                        <span>${client.name}</span>
                        <small>ID: ${client.id} | Progreso: ${client.services}/10</small>
                    </div>
                    <div style="display:flex; gap:6px;">
                        <button class="btn-small" onclick="selectClient(${realIndex})">Ver</button>
                        <button class="btn-small secondary" style="background-color:var(--bg-main); color:var(--sec-1);" onclick="requestAdminAction('edit', ${realIndex})">Editar</button>
                    </div>
                `;
                container.appendChild(item);
            });
        }

        function clearSearch() {
            document.getElementById('searchInput').value = '';
            filterClients();
        }

        function selectClient(index) {
            currentClientIndex = index;
            renderCard();
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function editClientName(index) {
            const client = clients[index];
            const newName = prompt('Editar nombre de la clienta:', client.name);
            if (newName !== null && newName.trim() !== '') {
                client.name = newName.trim();
                saveData();
                renderCard();
                filterClients();
            }
        }

        // Inicializar app al cargar
        renderCard();
    </script>
